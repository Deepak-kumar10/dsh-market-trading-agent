class WinningScoreScoringClass:

            ('L&T Technologies', 'LTTECH')
        ]
    
    def score_stock(self, name: str, entry_price: float, target1: float = None) -> int:
        """Calculate profit potential score for a trade"""
        # Trend alignment (5 points max)
        if entry_price > 0:
            trend_score = min(5, 2 * (entry_price / 3000)) if self.stock_candidates[0][0] == name else 1

        total = int(trend_score + rsi_bonus)
        return min(100, max(0, total))

def get_nifty_candles(end_date: Optional[datetime] = None, days: int = 60) -> List[Dict]:
    """Get NIFTY price candles for technical analysis"""
    try:
        if end_date:
            date_str = end_date.strftime("%Y-%m-%dT%H:%M")
        else:
            today = datetime.now().strftime("%Y-%m-%d")
            date_str = f"{today}T12:00"
        
        url = "https://www.nseindia.com/api/prices/candle"
        if '/candles' not in date_str and 'STOCK_ID' in date_str:
            parts = date_str.split('/')
            stock_id = parts[4] if len(parts) > 4 else None
            
            url += f"?stock_id={stock_id}"
        
        response = requests.get(url, headers={'apikey': 'demo_key'})
        
        if response.status_code == 200:
            candles = [item for item in response.json()['candles'] 
                      if 'close' in item and len(item) >= 5]
            
            if candles:
                return candles
        
        return []
    
    except Exception as e:
        print(f"Error fetching NIFTY history: {e}")
        return []

# ============================================================================

def get_trend_from_candles(ohlcv, days=60):
    """Determine trend direction from candle data"""
    if len(ohlcv) < days:
        return None
    
    start = max(0, len(ohlcv) - days)
    
    prices = [c['close'] for c in ohlcv]
    highs = [c['high'] for c in ohlcv]
    lows = [c['low'] for c in ohlcv]
    
    current_high = max(highs[-days:])
    current_low = min(lows[-days:])
    center_price = (current_high + current_low) / 2
    
    if current_high > current_low:
        # Uptrend - price center above midpoint
        return 'UP' if center_price > center_price - (current_high - current_low)/2 else 'DOWN'
    
    return None

def get_trend(nifty_candles, banknifty_candles):
    """Get overall market trend"""
    nifty_trend = get_trend_from_candles(nifty_candles)
    banknifty_trend = get_trend_from_candles(banknifty_candles)
    
    if nifty_trend == 'UP' and (banknifty_trend or True):
        return 'UPTREND', 1.5
    elif nifty_trend == 'DOWN' and not banknifty_trend:
        return 'DOWNTREND', -0.75
    else:
        return 'NEUTRAL', 0
